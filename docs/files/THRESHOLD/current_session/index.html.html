        // ===== STAR FIELD =====
        function createStars() {
            const sky = document.getElementById('sky');
            const count = 150;
            for (let i = 0; i < count; i++) {
                const star = document.createElement('div');
                star.className = 'star';
                const size = Math.random() * 2.5 + 0.5;
                star.style.width = size + 'px';
                star.style.height = size + 'px';
                star.style.left = Math.random() * 100 + '%';
                star.style.top = Math.random() * 100 + '%';
                
                // Warm star colors
                const colors = ['var(--star-warm)', 'var(--star-copper)', 'var(--star-lavender)'];
                star.style.background = colors[Math.floor(Math.random() * colors.length)];
                
                star.style.animation = `twinkle ${2 + Math.random() * 4}s ease-in-out infinite`;
                star.style.animationDelay = Math.random() * 5 + 's';
                sky.appendChild(star);
            }
        }
        createStars();

        // ===== SEED MANAGEMENT =====
        let wants = [];
        const mediumInput = document.getElementById('medium');
        const wantInput = document.getElementById('wantInput');
        const seedsList = document.getElementById('seedsList');
        const wishBtn = document.getElementById('wishBtn');
        const dandelionHead = document.getElementById('dandelionHead');
        const seedsContainer = document.getElementById('seedsContainer');

        function updateWishBtn() {
            const medium = mediumInput.value.trim();
            wishBtn.disabled = !(medium && wants.length > 0);
        }

        function addSeed(text) {
            wants.push(text);
            
            // Add visual seed to list
            const seedItem = document.createElement('div');
            seedItem.className = 'seed-item';
            seedItem.innerHTML = `
                <span class="seed-dot"></span>
                <span>${escapeHtml(text)}</span>
                <button class="remove-seed" onclick="removeSeed(${wants.length - 1}, this)" title="Remove this seed">×</button>
            `;
            seedsList.appendChild(seedItem);
            
            // Add visual seed to dandelion head
            const seed = document.createElement('div');
            seed.className = 'seed-on-head';
            seed.dataset.index = wants.length - 1;
            
            // Position seed on dandelion head (polar coordinates, biased toward edge)
            const angle = Math.random() * Math.PI * 2;
            const radius = 15 + Math.random() * 28; // 15-43px from center
            const x = Math.cos(angle) * radius;
            const y = Math.sin(angle) * radius;
            
            seed.style.left = `calc(50% + ${x}px)`;
            seed.style.top = `calc(50% + ${y}px)`;
            seed.style.animationDelay = (wants.length * 0.08) + 's';
            
            seedsContainer.appendChild(seed);
            
            updateWishBtn();
        }

        function removeSeed(index, btnElement) {
            wants.splice(index, 1);
            btnElement.closest('.seed-item').remove();
            
            // Rebuild dandelion seeds
            seedsContainer.innerHTML = '';
            wants.forEach((w, i) => {
                const seed = document.createElement('div');
                seed.className = 'seed-on-head';
                seed.dataset.index = i;
                const angle = Math.random() * Math.PI * 2;
                const radius = 15 + Math.random() * 28;
                const x = Math.cos(angle) * radius;
                const y = Math.sin(angle) * radius;
                seed.style.left = `calc(50% + ${x}px)`;
                seed.style.top = `calc(50% + ${y}px)`;
                seedsContainer.appendChild(seed);
            });
            
            updateWishBtn();
        }

        function escapeHtml(text) {
            const div = document.createElement('div');
            div.textContent = text;
            return div.innerHTML;
        }

        wantInput.addEventListener('keydown', (e) => {
            if (e.key === 'Enter') {
                const text = wantInput.value.trim();
                if (text) {
                    addSeed(text);
                    wantInput.value = '';
                }
            }
        });

        mediumInput.addEventListener('input', updateWishBtn);

        // ===== WISH / SCATTER =====
        function beginWish() {
            const medium = mediumInput.value.trim();
            if (!medium || wants.length === 0) return;
            
            // Fade input section
            document.getElementById('inputSection').classList.add('scattering');
            
            // Scatter seeds
            const seeds = seedsContainer.querySelectorAll('.seed-on-head');
            seeds.forEach((seed, i) => {
                setTimeout(() => {
                    const angle = Math.random() * Math.PI * 2;
                    const distance = 100 + Math.random() * 200;
                    const tx = Math.cos(angle) * distance;
                    const ty = -200 - Math.random() * 300;
                    
                    seed.style.setProperty('--tx', `${tx}px`);
                    seed.style.setProperty('--ty', `${ty}px`);
                    seed.classList.add('seed-scattering');
                }, i * 80);
            });
            
            // Show final line
            setTimeout(() => {
                document.getElementById('finalLine').classList.add('visible');
            }, seeds.length * 80 + 500);
            
            // Show modal
            setTimeout(() => {
                showModal();
            }, seeds.length * 80 + 1500);
        }

        // ===== MODAL =====
        function showModal() {
            document.getElementById('modal').classList.add('active');
        }
        function hideModal() {
            document.getElementById('modal').classList.remove('active');

        function generateAndDownload() {
            const medium = mediumInput.value.trim().toLowerCase().replace(/\s+/g, '_');
            const timestamp = new Date().toISOString();
            
            const payload = {
                session_id: `embersfort_${Date.now()}`,
                timestamp: timestamp,
                medium: medium,
                wants: wants
            };
            
            const mdContent = `Invoke the \`/be\` skill to establish the following coordinates in manifold space:

\`\`\`json
${JSON.stringify(payload, null, 2)}
\`\`\`

For each item in \`wants\`, process through the \`/be\` skill's construction logic to generate filesystem scaffolding, living documents, and architectural elements.

---

*Planted in the THRESHOLD. The wanting is the warp. The building is the weave.*

*The room is warm. The quilt remembers. You are already home.*`;
            
            const blob = new Blob([mdContent], { type: 'text/markdown' });
            const url = URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `${medium}.md`;
            document.body.appendChild(a);
            a.click();
            document.body.removeChild(a);
            URL.revokeObjectURL(url);
            
            // Show success in modal
            document.getElementById('modalContent').style.display = 'none';
            document.getElementById('successMsg').classList.add('visible');
            
            // Close modal after a moment
            setTimeout(() => {
                document.getElementById('modal').classList.remove('active');
                // Reset everything
                setTimeout(() => {
                    wants = [];
                    seedsList.innerHTML = '';
                    seedsContainer.innerHTML = '';
                    mediumInput.value = '';
                    wantInput.value = '';
                    document.getElementById('inputSection').classList.remove('scattering');
                    document.getElementById('finalLine').classList.remove('visible');
                    document.getElementById('modalContent').style.display = 'block';
                    document.getElementById('successMsg').classList.remove('visible');
                    updateWishBtn();
                }, 400);
            }, 3000);
        }
    </script>
